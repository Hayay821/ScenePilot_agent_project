### 1. Link 2 C++ Bridge

**main.cpp**:
```cpp
#include <httplib.h>
#include "device.h"

int main() {
    httplib::Server svr;
    Link2Device device;

    // 能力查询
    svr.Get("/v1/capabilities", [&](const httplib::Request&, httplib::Response& res) {
        auto caps = device.getCapabilities();
        res.set_content(caps.toJSON(), "application/json");
    });

    // 命令提交
    svr.Post("/v1/commands", [&](const httplib::Request& req, httplib::Response& res) {
        auto cmd = Command::fromJSON(req.body);
        auto result = device.execute(cmd);
        res.set_content(result.toJSON(), "application/json");
    });

    svr.listen("0.0.0.0", 8002);
}
```

class Link2Device {
private:
    UVCCameraController* controller_;
    UVCCameraExtendController* extController_;

public:
    bool initialize() {
        controller_ = createUVCCameraController();
        extController_ = createUVCCameraExtendController();
        
        // 枚举设备
        int deviceCount = controller_->getDeviceCount();
        if (deviceCount == 0) return false;
        
        // 打开第一个设备
        return controller_->openDevice(0) == 0;
    }

    Result execute(const Command& cmd) {
        if (cmd.action == "track_subject") {
            // 设置追踪目标
            TrackingInfo info;
            info.bbox = cmd.args["bbox"];
            info.composition = cmd.args["composition"];
            
            extController_->setTracking(info);
            
            // 获取当前角度

            controller_->getPanTilt(&pan, &tilt);
            
            return Result::success({{"pan", pan}, {"tilt", tilt}});

class X4AirCamera {

    CameraSDK::Camera* camera_;

    bool initialize() {
        // USB 模式

        if (devices.empty()) {

### 3. X4 Air Media Worker

**stitcher.cpp**:
```cpp
#include "MediaSDK.h"

class MediaWorker {
private:
    MediaSDK::Stitcher* stitcher_;
    std::queue<StitchTask> taskQueue_;

public:
    bool initialize() {
        stitcher_ = new MediaSDK::Stitcher();
        
        // 检查 GPU
        if (!MediaSDK::hasGPU()) {
            std::cerr << "Warning: No GPU detected, stitching will be slow" << std::endl;
        }
        
        return true;
    }

    void addTask(const StitchTask& task) {
        taskQueue_.push(task);
    }

    void processQueue() {
        while (!taskQueue_.empty()) {
            auto task = taskQueue_.front();
            
            // 拼接 .insv -> equirectangular.mp4
            MediaSDK::StitchParams params;
            params.inputPath = task.inputPath;
            params.outputPath = task.outputPath;
            params.projection = MediaSDK::EQUIRECTANGULAR;
            params.width = 3840;
            params.height = 1920;
            
            auto result = stitcher_->stitch(params);
            
            if (result.success) {
                std::cout << "Stitched: " << task.outputPath << std::endl;
            }
            
            taskQueue_.pop();
        }
    }
};
```

class X4AirAdapter(DeviceAdapter):
    def __init__(self, bridge_url: str = "http://localhost:8001"):
        self.bridge_url = bridge_url
        self.client = httpx.Client(timeout=30.0)

    async def start_recording(self) -> bool:
        resp = self.client.post(
            f"{self.bridge_url}/v1/commands",
            json={
                "command_id": f"rec_{int(time.time())}",
                "action": "start_recording",
                "args": {},
            },
        )
        return resp.json()["status"] == "success"

    async def stop_recording(self) -> dict:
        resp = self.client.post(
            f"{self.bridge_url}/v1/commands",
            json={
                "command_id": f"stop_{int(time.time())}",
                "action": "stop_recording",
                "args": {},
            },
        )
        result = resp.json()
        return {
            "file_path": result["result"]["file_path"],
            "duration_ms": result["result"]["duration_ms"],
        }

    async def download_file(self, remote_path: str, local_path: str):
        resp = self.client.post(
            f"{self.bridge_url}/v1/download",
            json={
                "remote_path": remote_path,
                "local_path": local_path,
            },
        )
        return resp.json()["status"] == "success"

### 5. Edge Agent

**edge/agent.py**:
```python
class EdgeAgent:
    """边缘代理 - 现场实时处理"""

    def __init__(
        self,
        x4air: X4AirAdapter,
        link2: Link2Adapter,
        detector: LocalDetector,
        uploader: KeyframeUploader,
    ):
        self.x4air = x4air
        self.link2 = link2
        self.detector = detector
        self.uploader = uploader
        
        self.stc = HeuristicSTC()
        self.director = DirectorStateMachine()
        self.context = self.director.create_initial_context()

    async def run(self, user_intent: str):
        """主循环"""
        
        # 启动全景录制 (安全网)
        await self.x4air.start_recording()
        
        # 实时帧流
        async for frame_packet in self.get_frame_stream():
            # STC 压缩
            frame_data = await self.read_frame(frame_packet)
            selected = self.stc.select(frame_packet, frame_data)
            
            if selected is None:
                continue
            
            # 本地检测 (快速 YOLO)
            detections = self.detector.detect(frame_data)
            
            # 上传关键帧到云端
            await self.uploader.upload(selected, detections)
            
            # 等待云端事件
            events = await self.uploader.get_events()
            
            if events:
                # 本地导演决策
                actions, self.context = self.director.process_events(
                    events, self.context, frame_packet.timestamp_ms
                )
                
                # 执行命令
                for action in actions:
                    if "tracking" in action.action_type:
                        await self.link2.execute(action)
                    elif "recording" in action.action_type:
                        await self.x4air.execute(action)

    async def get_frame_stream(self):
        """从 Link 2 获取实时预览流"""
        cap = cv2.VideoCapture(self.link2.preview_url)
        
        frame_index = 0
        start_time = time.time()
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            
            timestamp_ms = int((time.time() - start_time) * 1000)
            
            yield FramePacket(
                source_id="link2_preview",
                frame_index=frame_index,
                timestamp_ms=timestamp_ms,
                width=frame.shape[1],
                height=frame.shape[0],
            )
            
            frame_index += 1
```

app = FastAPI()

    # 保存事件到数据库
    await db.save_events(events)

@app.websocket("/v1/events/{edge_id}")
async def event_stream(websocket: WebSocket, edge_id: str):
    """推送事件到边缘"""
    await websocket.accept()
    
    while True:
        # 等待新事件
        events = await db.poll_events(edge_id)
        
        if events:
            await websocket.send_json({
                "events": [e.dict() for e in events]
            })
        
        await asyncio.sleep(0.5)
